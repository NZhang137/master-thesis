\chapter{Introduction}

This thesis studies preference-aware coefficient correction for model merging under a fixed interpolation family. The central question is whether a geometry-aware mapping
\[
\lambda=f(p,R)
\]
can improve coefficient selection relative to the direct baseline \(\lambda=p\), while staying in the same Rewarded-Soups-style family.

This thesis is organized around three research questions; \textbf{RQ2} is the pivotal one, since it governs whether the others can be answered affirmatively.

\begin{description}[leftmargin=2.2em,style=nextline]
\item[\textbf{RQ1 (Does geometry-aware correction help?)}]
Within the fixed Rewarded-Soups interpolation family \(\mathcal{M}=\{\theta(\lambda):\lambda\in\Delta_m\}\), can a conflict-aware mapping \(\lambda=f(p,R)\) achieve higher preference-weighted utility \(U_p\) than the direct baseline \(\lambda=p\), and than search-based selection at a matched compute budget?
\item[\textbf{RQ2 (Is \(R\) a valid proxy, and when does it help?)}]
To what extent does the parameter-geometric relationship matrix \(R\) predict reward-space behaviour? Because \(R\) never observes the rewards directly, quantifying the strength of this proxy --- and the regimes in which it holds --- is what determines when \(f(p,R)\) can be expected to help, and whether reward-informed correction is needed to close any remaining gap.
\item[\textbf{RQ3 (How does the benefit scale with \(m\)?)}]
Dense search over the simplex \(\Delta_m\) becomes expensive as the number of objectives \(m\) grows. How does the advantage of a one-shot mapping \(f(p,R)\) over budgeted search scale with \(m\)?
\end{description}

\chapter{Background and Related Work}

Rewarded Soups motivates the fixed linear interpolation family considered here \citep{rame2023rewarded}. Task Arithmetic motivates the use of task vectors defined relative to a shared base model \citep{ilharco2022taskarithmetic}. TIES and AdaMerging are nearby model-merging baselines \citep{yadav2023ties,yang2023adamerging}. MGDA, PCGrad, and CAGrad are gradient-based multi-objective methods that here serve only as inspiration for one-shot, \(R\)-based coefficient mappings rather than their original training-time algorithms \citep{sener2018mgda,yu2020pcgrad,liu2021cagrad}. LoRA is the practical adapter mechanism for the prototype implementation \citep{hu2021lora}.

\section{Multi-objective methods that inspire the coefficient mappings}

For context, we briefly summarize the three multi-objective optimization methods that inspire the coefficient mappings defined in Chapter~\ref{chap:method}. In all three cases only the underlying geometric principle is reused; it is applied to the static relationship matrix \(R\) rather than to live per-step gradients, and no coefficient-space gradient descent is performed.

\emph{MGDA (Multiple Gradient Descent Algorithm)} casts multi-task learning as multi-objective optimization \citep{sener2018mgda}. Given a set of per-task gradients, it selects the convex combination of minimum aggregated norm; the resulting update direction lies in the common descent cone, so no individual task loss increases to first order. Strongly conflicting tasks pull the combination weights toward a balancing point.

\emph{PCGrad (Gradient Surgery)} resolves conflicts locally and pairwise \citep{yu2020pcgrad}. Whenever two task gradients have a negative inner product (an interference), one gradient is projected onto the orthogonal complement of the other, removing only the conflicting component while leaving complementary directions intact. This contrasts with MGDA, which computes a single global balance.

\emph{CAGrad (Conflict-Averse Gradient Descent)} optimizes the average objective while constraining the update to a trust region around the average gradient \citep{liu2021cagrad}. The constraint bounds the worst-case per-task decrease, trading off average progress against worst-case protection.

In this thesis these ideas are transferred to one-shot coefficient selection: the gradient Gram matrix is replaced by the static relationship matrix \(R\), conflict projections and trust regions are expressed in the \(R\)-induced geometry, and the output is a single coefficient vector \(\lambda\in\Delta_m\) rather than an iterative descent trajectory.

\section{Objectives and rewards}

\begin{definition}[Objectives and rewards]
Let \(\Theta:=\R^d\) denote the parameter space. For each objective \(i\in\{1,\dots,m\}\), let
\[
r_i:\Theta\to\R
\]
be a reward function. Define
\[
r(\theta):=\bigl(r_1(\theta),\dots,r_m(\theta)\bigr)\in\R^m.
\]
If a loss notation is useful, define optionally
\[
F_i(\theta):=-\,r_i(\theta).
\]
This thesis does \emph{not} define coefficient-space losses such as \(F_i(\lambda)\).
\end{definition}

\begin{definition}[Preference vector]
The user preference vector is
\[
p\in\Delta_m,\qquad
\Delta_m:=\{x\in\R^m:x_i\ge 0\ \forall i,\ \1^\top x=1\}.
\]
\end{definition}

\begin{definition}[Preference-weighted utility]
For fixed \(p\in\Delta_m\), define
\[
U_p(\theta):=\sum_{i=1}^m p_i\,r_i(\theta).
\]
If reward scales differ, let \(\tilde r_i(\theta)\) denote fixed normalized rewards and evaluate with
\[
U_p(\theta):=\sum_{i=1}^m p_i\,\tilde r_i(\theta).
\]
\end{definition}

\begin{remark}[Normalization]
If normalized rewards are used, the normalization rule should be fixed before comparing coefficient-selection methods.
\end{remark}

\section{Base model, specialists, and task vectors}

\begin{assumption}[Shared-base requirement]
There exists a shared base model \(\theta_0\in\Theta\). For each objective \(i\in\{1,\dots,m\}\), the specialist model \(\theta_i\in\Theta\) is obtained from the same base model \(\theta_0\), with the same parameterization and parameter ordering.
\end{assumption}

\begin{definition}[Task vectors]
For each specialist \(i\), define
\[
\delta_i:=\theta_i-\theta_0\in\R^d.
\]
The shared-base requirement is necessary for the task-vector interpretation and the linear interpolation family used in this thesis.
\end{definition}

\begin{definition}[Rewarded-Soups-style interpolation family]
For \(\lambda\in\Delta_m\), define
\[
\theta(\lambda):=\theta_0+\sum_{i=1}^m \lambda_i\delta_i.
\]
Equivalently, because \(\sum_i\lambda_i=1\) and \(\delta_i=\theta_i-\theta_0\),
\[
\theta(\lambda)=\sum_{i=1}^m \lambda_i\theta_i.
\]
Define the fixed interpolation family
\[
\mathcal{M}:=\{\theta(\lambda):\lambda\in\Delta_m\}.
\]
This thesis operates only inside \(\mathcal{M}\).
\end{definition}

\begin{remark}[Non-claim]
Any empirical gain discussed later would concern only \emph{selection of \(\lambda\) inside \(\mathcal{M}\)}. It would not imply a global Pareto-front improvement over all \(\theta\in\Theta\).
\end{remark}

\section{Relationship matrix from task-vector geometry}

\begin{definition}[Task-vector matrix]
Let
\[
D:=[\delta_1,\dots,\delta_m]\in\R^{d\times m}.
\]
Then \(\theta(\lambda)=\theta_0+D\lambda\).
\end{definition}

\begin{definition}[Relationship matrix]
A static relationship matrix \(R\in\R^{m\times m}\) can be defined in two canonical ways:

\[
R:=D^\top D,\qquad R_{ij}=\delta_i^\top\delta_j,
\]
or, if \(\delta_i\neq 0\) for all \(i\), through normalized task vectors
\[
\hat\delta_i:=\frac{\delta_i}{\norm{\delta_i}_2},\qquad \hat D:=[\hat\delta_1,\dots,\hat\delta_m],\qquad R:=\hat D^\top\hat D,
\]
so that
\[
R_{ij}=\cos(\delta_i,\delta_j)=\frac{\delta_i^\top\delta_j}{\norm{\delta_i}_2\norm{\delta_j}_2}.
\]

Interpretation:
\begin{itemize}
    \item positive values may indicate alignment or complementarity,
    \item negative values may indicate conflict or suppression,
    \item high positive similarity may also indicate redundancy.
\end{itemize}

This remains only a parameter-geometric proxy and must be empirically validated.
\end{definition}

\begin{remark}[PSD status]
If \(R=D^\top D\), then \(R\) is positive semidefinite. If cosine similarity is implemented as the normalized Gram matrix \(R=\hat D^\top\hat D\), then it is also positive semidefinite. 
%If one uses a non-Gram similarity table, PSD is not guaranteed.
\end{remark}

\section{Preference-to-coefficient mapping}

\begin{definition}[Preference-to-coefficient mapping]
The main object of study is
\[
f:\Delta_m\times\R^{m\times m}\to\Delta_m,\qquad \lambda=f(p,R).
\]
The direct preference baseline is
\[
\lambda_{\mathrm{direct}}:=p.
\]
Even though \(p\) and \(\lambda\) lie in the same simplex, they are conceptually different:
\begin{itemize}
    \item \(p\) is a preference vector in reward/objective space,
    \item \(\lambda\) is a merge coefficient vector in parameter/model space.
\end{itemize}
\end{definition}

\begin{definition}[Coefficient-selection problem]
Given \(p\in\Delta_m\) and a fixed relationship matrix \(R\), the thesis studies whether a geometry-aware correction
\[
\lambda=f(p,R)
\]
can improve coefficient selection relative to \(\lambda=p\), while keeping \(\theta(\lambda)\) inside the same interpolation family \(\mathcal{M}\). The intended pipeline is
\[
(\delta_i)_{i=1}^m \longrightarrow R \longrightarrow \lambda=f(p,R) \longrightarrow \theta(\lambda).
\]
\end{definition}

\begin{remark}[Selection cost is independent of the parameter dimension]
Every coefficient mapping \(f(p,R)\) operates only on the relationship matrix \(R\in\R^{m\times m}\) and the coefficient vector \(\lambda\in\R^m\). The high-dimensional parameter space enters exactly once, when \(R=D^\top D\) is formed from the task vectors at cost \(O(m^2 d)\); this is computed a single time and reused for every preference. Thereafter, solving for \(\lambda=f(p,R)\) --- and re-solving for any new preference \(p\) --- costs \(\mathrm{poly}(m)\) and is \emph{independent of} \(d\). The parameter dimension \(d\) (millions, for LoRA adapters) is touched again only to realize the final merge \(\theta(\lambda)=\theta_0+D\lambda\) for a chosen \(\lambda\). By contrast, search over \(\mathcal{B}\) requires \(\lvert\mathcal{B}\rvert\) separate merges and reward evaluations, each touching \(d\). This is the precise sense in which the proposed mappings are one-shot and cheap: the selection problem is solved entirely in the \(m\)-dimensional space.
\end{remark}

\section{Baselines and evaluation}

\begin{definition}[Baselines]
The following baselines are considered:
\[
\lambda_i^{\mathrm{unif}}=\frac{1}{m},\qquad
\lambda^{\mathrm{direct}}=p.
\]
In addition, one may use random search, Dirichlet sampling, or Sobol-style search over \(\Delta_m\).
\end{definition}

\begin{remark}[Finite-search notation]
To avoid a clash with LoRA matrices \(B_i\), the finite evaluated coefficient set is denoted by \(\mathcal{B}\subset\Delta_m\).
\end{remark}

\begin{definition}[High-budget best-found reference]
Given finite \(\mathcal{B}\subset\Delta_m\), define
\[
\lambda_{\mathrm{best}}(p):=\argmax_{\lambda\in\mathcal{B}}U_p(\theta(\lambda)).
\]
This is a finite-search reference, not a true oracle over all of \(\Delta_m\).
\end{definition}

\begin{definition}[Evaluation metrics]
All evaluation metrics are defined for coefficient vectors \(\lambda\in\Delta_m\) and therefore evaluate only models inside the fixed family
\[
\mathcal{M}=\{\theta(\lambda):\lambda\in\Delta_m\}.
\]
They are not global Pareto-front metrics.

Let \(\tilde r_i(\theta(\lambda))\) denote the evaluation reward for objective \(i\). If objective scales are already comparable, we set \(\tilde r_i=r_i\). Otherwise, the normalization rule is fixed before comparing methods. A typical min--max normalization uses constants \(a_i<b_i\), obtained from known reward ranges or from a separate calibration set, and defines
\[
\tilde r_i(\theta):=\frac{r_i(\theta)-a_i}{b_i-a_i}.
\]
If the normalization constants are instead estimated from the finite evaluated set \(\mathcal{B}\), this choice must be reported explicitly because the same set is also used to define the best-found reference.

For a fixed preference vector \(p\in\Delta_m\), the individual objective scores are
\[
\tilde r_i(\theta(\lambda)),\qquad i=1,\dots,m.
\]
%The average reward is the unweighted diagnostic
%\[
%\operatorname{AvgReward}(\lambda):=\frac{1}{m}\sum_{i=1}^m %\tilde r_i(\theta(\lambda)),
%\]
%where larger values are better. 
The preference-weighted utility is
\[
U_p(\theta(\lambda)):=\sum_{i=1}^m p_i\,\tilde r_i(\theta(\lambda)),
\]
where larger values are better for the fixed preference \(p\).

The direct preference baseline is \(\lambda=p\). To avoid confusing the preference vector with a model parameter, we write the baseline model explicitly as \(\theta(\lambda=p)\). The utility improvement over direct preference interpolation is
\[
\Delta U_p(\lambda_{\mathrm{method}}):=
U_p(\theta(\lambda_{\mathrm{method}}))-U_p(\theta(\lambda=p)).
\]
Thus \(\Delta U_p>0\) means that the method improves over the direct baseline for the same preference \(p\).

The best-found reference is computed separately for each preference:
\[
\lambda_{\mathrm{best}}^{\mathcal{B}}(p):=
\argmax_{\lambda\in\mathcal{B}}U_p(\theta(\lambda)).
\]
The superscript \(\mathcal{B}\) emphasizes that this is only the best point found in the finite evaluated set. The within-family utility gap is
\[
\operatorname{Gap}_p^{\mathcal{B}}(\lambda_{\mathrm{method}}):=
U_p\bigl(\theta(\lambda_{\mathrm{best}}^{\mathcal{B}}(p))\bigr)
-U_p(\theta(\lambda_{\mathrm{method}})).
\]
A smaller gap is better. If \(\lambda_{\mathrm{method}}\in\mathcal{B}\), then \(\operatorname{Gap}_p^{\mathcal{B}}(\lambda_{\mathrm{method}})\ge 0\).

Preference faithfulness is reported as a diagnostic trade-off, not as a reward metric. It can be measured by
\[
\norm{\lambda_{\mathrm{method}}-p}_1
\qquad \text{or}\qquad
\norm{\lambda_{\mathrm{method}}-p}_2.
\]
An optional geometry-aware diagnostic is
\[
D_{\bar R}(\lambda_{\mathrm{method}},p):=
(\lambda_{\mathrm{method}}-p)^\top \bar R(\lambda_{\mathrm{method}}-p),
\]
where \(\bar R\succeq 0\) is the matrix used for distance evaluation. If \(R\) is already a Gram matrix, for example \(R=D^\top D\) or \(R=\hat D^\top\hat D\), we may set \(\bar R=R\). If \(R\) is not guaranteed to be symmetric positive semidefinite, then \(\bar R\) should be obtained by symmetrizing and projecting to the positive semidefinite cone. If \(\bar R=D^\top D\), then
\[
D_{\bar R}(\lambda_{\mathrm{method}},p)=
\norm{D(\lambda_{\mathrm{method}}-p)}_2^2.
\]

For a Tchebychev-style worst-case shortfall metric, define observed best and worst normalized rewards over the finite evaluated set
\[
z_i^{\mathrm{best}}:=\max_{\lambda\in\mathcal{B}}\tilde r_i(\theta(\lambda)),
\qquad
z_i^{\mathrm{worst}}:=\min_{\lambda\in\mathcal{B}}\tilde r_i(\theta(\lambda)).
\]
For numerical stability, let \(\varepsilon_T>0\). Define the active objective set
\[
A_{\varepsilon_p}(p):=\{i\in\{1,\dots,m\}:p_i>\varepsilon_p\},
\]
with a small threshold \(\varepsilon_p\ge 0\). The normalized Tchebychev-style metric is
\[
T_p^{\mathrm{norm}}(\lambda):=
\max_{i\in A_{\varepsilon_p}(p)}
p_i\left[
\frac{z_i^{\mathrm{best}}-\tilde r_i(\theta(\lambda))}
{z_i^{\mathrm{best}}-z_i^{\mathrm{worst}}+\varepsilon_T}
\right].
\]
Smaller values are better. This convention multiplies the normalized shortfall by \(p_i\), so shortfalls in high-preference objectives are penalized more strongly. Objectives with zero preference do not affect this metric. Objectives with negligible observed range should be handled carefully because the denominator can amplify small absolute differences.

The Tchebychev improvement over the direct baseline is
\[
\Delta T_p^{\mathrm{norm}}(\lambda_{\mathrm{method}}):=
T_p^{\mathrm{norm}}(\lambda=p)-T_p^{\mathrm{norm}}(\lambda_{\mathrm{method}}).
\]
A positive value means that the method reduces the worst-case preference-weighted shortfall relative to \(\lambda=p\).

Finally, evaluation is performed over a set of preference vectors
\[
\mathcal{P}:=\{p^{(1)},\dots,p^{(K)}\}\subset\Delta_m.
\]
Metrics such as \(\Delta U_p\), \(\operatorname{Gap}_p^{\mathcal{B}}\), \(T_p^{\mathrm{norm}}\), and preference-faithfulness distances are reported per preference and summarized over \(\mathcal{P}\) using mean, median, and robust statistics. 
%Additional useful diagnostics include the win rate over the direct baseline,
%\[
%\frac{1}{K}\sum_{k=1}^K
%\mathbf{1}\!\left\{
%U_{p^{(k)}}(\theta(\lambda_{\mathrm{method}}^{(k)}))>
%U_{p^{(k)}}(\theta(\lambda=p^{(k)}))
%\right\},
%\]
%and the finite-set rank of \(\lambda_{\mathrm{method}}\) according to \(U_p\) among the evaluated coefficients in \(\mathcal{B}\).
\end{definition}

\begin{remark}[Tchebychev weighting convention]
The metric \(T_p^{\mathrm{norm}}\) follows the standard weighted Tchebycheff scalarization \citep{miettinen1999nonlinear}, in which the weight \emph{multiplies} the deviation from the ideal reference point,
\[
g_{\mathrm{tch}}(\theta\mid w,z^\star)=\max_{i=1,\dots,m}\bigl\{\,w_i\,\lvert f_i(\theta)-z_i^\star\rvert\,\bigr\}.
\]
Choosing \(w_i=p_i\) is motivated by three considerations. First, preference consistency: a shortfall on a high-preference objective should be penalized more than one on a low-preference objective. Second, consistency with the primary metric \(U_p\), which also weights rewards by \(p_i\); both the average-case and the worst-case metric then share the same preference perspective. Third, the zero-preference case is handled cleanly, since objectives with \(p_i\le\varepsilon_p\) are excluded through \(A_{\varepsilon_p}(p)\), whereas an inverse \(1/p_i\) weighting would diverge there. The inverse convention \(w_i\propto 1/p_i\) arises only when scalarization is used to \emph{place} a solution along a target direction \(p\) on the Pareto front, as in MORL/MOEA/D decomposition; this is a solution-selection technique rather than an evaluation principle, and is not used here.
\end{remark}

\begin{remark}[Evaluation scope]
All quantities above compare only coefficient choices inside the fixed family \(\mathcal{M}\) or the finite evaluated set \(\mathcal{B}\). The notation \(\lambda_{\mathrm{best}}^{\mathcal{B}}(p)\) should not be interpreted as a global optimum, and improvements in \(\Delta U_p\) or \(\Delta T_p^{\mathrm{norm}}\) do not imply global Pareto-front improvement.
\end{remark}

\begin{remark}[A negative \(\operatorname{Gap}_p^{\mathcal{B}}\) is a success, not an anomaly]
The reference \(\lambda_{\mathrm{best}}^{\mathcal{B}}(p)\) is the best coefficient within a \emph{finite} search set \(\mathcal{B}\) of budget \(\lvert\mathcal{B}\rvert\), not a ground-truth optimum over \(\Delta_m\). Consequently \(\operatorname{Gap}_p^{\mathcal{B}}\) can be negative: this happens exactly when a method finds a coefficient that no point in \(\mathcal{B}\) matches, i.e.\ when \(f(p,R)\) outperforms budgeted search. A negative \(\operatorname{Gap}_p^{\mathcal{B}}\) should therefore be read as evidence for the efficiency claim of RQ3, and \(\operatorname{Gap}_p^{\mathcal{B}}\) should always be reported together with the budget \(\lvert\mathcal{B}\rvert\) that defines the reference.
\end{remark}


\chapter{Method}
\label{chap:method}

This chapter defines one-shot mappings \(\lambda=f(p,R)\) that use only the preference vector \(p\) and a static relationship matrix \(R\). No coefficient-space gradients are introduced. We define six mappings, grouped by the multi-objective method that inspires them: \emph{M1} and \emph{M2} (MGDA-inspired), \emph{P1} and \emph{P2} (PCGrad / conflict-inspired), and \emph{C1} and \emph{C2} (CAGrad-inspired).

\section{General setup}

For optimization-based methods below, assume \(R\) is symmetric and positive semidefinite, e.g. \(R=D^\top D\) or \(R=\hat D^\top\hat D\). \todo{Decide later whether implementation uses raw \(R\), symmetrized \(R\), or a PSD approximation.}

\section{MGDA-inspired coefficient mapping}

\begin{definition}[M1: MGDA-inspired one-shot mapping]
Given \(p\in\Delta_m\), \(R\in\R^{m\times m}\), and \(\rho>0\), define
\[
\lambda_{\mathrm{M1}}
:=
\argmin_{\lambda\in\Delta_m}
\lambda^\top R\lambda+\rho\norm{\lambda-p}_2^2.
\]
This is inspired by min-norm balancing, but adapted to coefficient selection rather than gradient descent.
\end{definition}

\begin{remark}[Possible limitation]
Depending on how informative \(R\) is, this objective may encourage cancellation or move too strongly toward low-norm combinations.
\end{remark}

\begin{definition}[M2: Preference-weighted \(\alpha\)-MGDA variant]
Let
\[
G_p:=\Diag(p)\,R\,\Diag(p),\qquad u:=\frac{1}{m}\1.
\]
Define
\[
\alpha^\star
:=
\argmin_{\alpha\in\Delta_m}
\alpha^\top G_p\alpha+\rho\norm{\alpha-u}_2^2,
\]
and then
\[
\lambda_{\mathrm{M2}}:=
\frac{\alpha^\star\odot p}{\1^\top(\alpha^\star\odot p)}.
\]
This variant corrects \(p\) multiplicatively instead of replacing it outright. It is well defined for any \(p\neq 0\); objectives with \(p_i=0\) receive coefficient zero.
\end{definition}

\section{PCGrad-inspired coefficient mapping}

We consider two conflict-based mappings: a closed-form shrinkage (P1) that requires no solver, and a projection-based reconstruction (P2) that more directly mirrors PCGrad's gradient surgery.

\begin{definition}[P1: Conflict-weighted closed-form shrinkage]
Given \(p\in\Delta_m\), \(R\in\R^{m\times m}\), and \(\beta>0\), define the per-objective conflict load
\[
\kappa_i:=\sum_{j\neq i}\max(0,\,-R_{ij}),
\]
the shrinkage factors
\[
s_i:=\frac{1}{1+\beta\,\kappa_i},
\]
and the raw weights \(\tilde\lambda_i:=p_i\,s_i\). The coefficient vector is the simplex normalization
\[
\lambda_{\mathrm{P1}}:=\frac{\tilde\lambda}{\1^\top\tilde\lambda}.
\]
This is a closed-form mapping: objectives that conflict strongly with others (large \(\kappa_i\)) are down-weighted relative to their preference \(p_i\), while objectives that are aligned or complementary keep \(s_i\approx 1\).
\end{definition}

\begin{remark}[Scale of \(\beta\) and \(R\)]
The conflict load \(\kappa_i\) inherits the scale of \(R\). With a cosine-normalized \(R=\hat D^\top\hat D\), the off-diagonal entries lie in \([-1,1]\) and \(\kappa_i\in[0,m-1]\), so \(\beta\) has a stable, dimensionless interpretation. With \(R=D^\top D\), the magnitudes depend on the task-vector norms, and \(\beta\) must be tuned accordingly or \(R\) rescaled. Since \(s_i>0\) always and \(p\) has at least one positive entry on \(\Delta_m\), the vector \(\tilde\lambda\) has positive sum and the normalization is always well defined.
\end{remark}

\begin{definition}[P2: Task-vector PCGrad with simplex reconstruction]
Initialize preference-weighted task directions by
\[
d_i:=p_i\delta_i,\qquad i=1,\dots,m.
\]
Process conflicting pairs in a fixed deterministic order. Whenever a current pair satisfies
\[
d_i^\top d_j<0,
\]
update
\[
d_i\leftarrow d_i-\frac{d_i^\top d_j}{\norm{d_j}_2^2+\varepsilon}d_j,\qquad \varepsilon\ge 0.
\]
After all chosen projections, aggregate
\[
d_{\mathrm{PC}}:=\sum_{i=1}^m d_i.
\]
Since \(d_{\mathrm{PC}}\) need not correspond to a valid convex merge coefficient, reconstruct coefficients by solving
\[
\lambda_{\mathrm{P2}}
:=
\argmin_{\lambda\in\Delta_m}
\norm{D\lambda-d_{\mathrm{PC}}}_2^2+\rho\norm{\lambda-p}_2^2.
\]
\end{definition}

\begin{remark}[Equivalent \(R\)-metric implementation]
Let \(e_i\) be the \(i\)-th standard basis vector and
\[
c_i:=p_i e_i.
\]
Then \(Dc_i=p_i\delta_i=d_i\), and
\[
c_i^\top R c_j=(Dc_i)^\top(Dc_j).
\]
Thus projection can also be performed in the \(R\)-induced metric:
\[
c_i\leftarrow c_i-\frac{c_i^\top Rc_j}{c_j^\top Rc_j+\varepsilon}\,c_j.
\]
\end{remark}

\begin{remark}[Heuristic nature]
This PCGrad-inspired mapping is heuristic and may depend on projection order. A deterministic ordering by strongest negative conflict can be used for reproducibility.
\end{remark}

\section{CAGrad-inspired coefficient mapping}

\begin{definition}[C1: Trust-Region CAGrad-inspired method]
Let \(p\in\Delta_m\), let \(R\succeq 0\), let \(c\ge 0\), and let \(\varepsilon>0\). Define
\[
\lambda_{\mathrm{C1}}
:=
\argmax_{\lambda\in\Delta_m}\min_{i=1,\dots,m}(R\lambda)_i
\]
subject to
\[
(\lambda-p)^\top R(\lambda-p)\le c^2\max(p^\top Rp,\varepsilon).
\]

Equivalently, introduce \(t\in\R\) and solve
\[
\max_{\lambda\in\Delta_m,\ t} t
\]
subject to
\[
(R\lambda)_i\ge t\quad \forall i,
\qquad
(\lambda-p)^\top R(\lambda-p)\le c^2\max(p^\top Rp,\varepsilon).
\]
\end{definition}

\begin{remark}[Interpretation]
If \(R=D^\top D\), then
\[
(R\lambda)_i=\delta_i^\top D\lambda.
\]
Thus the objective protects against worst-case negative alignment in the proxy geometry while the trust region keeps \(\lambda\) close to \(p\). This is a one-shot coefficient mapping inspired by conflict-averse aggregation, not the original CAGrad training algorithm.
\end{remark}

\begin{definition}[C2: Soft-min CAGrad-inspired variant]
For \(\tau>0\), define
\[
\softmin_\tau(z):=-\tau\log\sum_{i=1}^m \exp(-z_i/\tau).
\]
Then
\[
\lambda_{\mathrm{C2}}
:=
\argmax_{\lambda\in\Delta_m}
\softmin_\tau(R\lambda)-\rho(\lambda-p)^\top R(\lambda-p).
\]
This is a smooth approximation of the worst-case alignment objective; as \(\tau\to 0^+\), \(\softmin_\tau\) converges to the minimum, recovering the C1 objective.
\end{definition}



\chapter{Practical LoRA-Based Implementation}

LoRA is a practical implementation choice here, not the theoretical contribution and not part of original Rewarded Soups.

\section{LoRA definition}

Let \(\mathcal{L}\) be the set of adapted linear layers. For layer \(\ell\in\mathcal{L}\), let \(W_\ell\) denote the frozen base weight. For objective \(i\), define
\[
W'_{i,\ell}=W_\ell+\Delta W_{i,\ell},
\qquad
\Delta W_{i,\ell}=B_{i,\ell}A_{i,\ell}.
\]
Optionally, with the usual scaling,
\[
\Delta W_{i,\ell}=\frac{\alpha_\ell}{r_\ell}B_{i,\ell}A_{i,\ell}.
\]

Define the effective multi-layer LoRA update
\[
\Delta_i^{\mathrm{LoRA}}:=(\Delta W_{i,\ell})_{\ell\in\mathcal{L}},
\]
and define the flattened LoRA task vector by
\[
\delta_i^{\mathrm{LoRA}}:=
\vecop\!\Bigl(\concat_{\ell\in\mathcal{L}}\Delta W_{i,\ell}\Bigr)\in\R^{d_{\mathrm{LoRA}}}.
\]

\section{LoRA geometry and merging}

Define the LoRA relationship matrix by either
\[
R_{ij}=(\delta_i^{\mathrm{LoRA}})^\top \delta_j^{\mathrm{LoRA}}
\]
or
\[
R_{ij}=
\cos(\delta_i^{\mathrm{LoRA}},\delta_j^{\mathrm{LoRA}})
=
\frac{(\delta_i^{\mathrm{LoRA}})^\top \delta_j^{\mathrm{LoRA}}}
{\norm{\delta_i^{\mathrm{LoRA}}}_2\norm{\delta_j^{\mathrm{LoRA}}}_2}.
\]

The effective updates need not be materialized as dense matrices. For two
adapters (i) and (j), their Frobenius inner product can be accumulated
layerwise as
\[
\left\langle \Delta W_{i,\ell},\Delta W_{j,\ell}\right\rangle_F
=
s_{i,\ell}s_{j,\ell}
\operatorname{tr}\!\left(
(B_{i,\ell}A_{i,\ell})^\top B_{j,\ell}A_{j,\ell}
\right),
\qquad
s_{i,\ell}:=\frac{\alpha_{i,\ell}}{r_{i,\ell}}.
\]
Equivalently, this quantity can be evaluated from the small factors as
\(s_{i,\ell}s_{j,\ell}\langle B_{i,\ell}^\top B_{j,\ell},
A_{i,\ell}A_{j,\ell}^\top\rangle_F\). Consequently,
the relationship matrix is invariant under any invertible factor
reparameterization \(B_{i,\ell}\mapsto B_{i,\ell}G\) and
\(A_{i,\ell}\mapsto G^{-1}A_{i,\ell}\) that leaves the effective update
unchanged. Raw concatenations of the \(A\)- and \(B\)-factors are therefore
not used to construct \(R\).

For \(\lambda\in\Delta_m\), define the merged effective LoRA update layerwise by
\[
\Delta W^{\mathrm{LoRA}}_{\mathrm{merged},\ell}
=
\sum_{i=1}^m \lambda_i \Delta W_{i,\ell}.
\]
Equivalently,
\[
\Delta^{\mathrm{LoRA}}_{\mathrm{merged}}
=
\sum_{i=1}^m \lambda_i \Delta_i^{\mathrm{LoRA}}.
\]

\begin{remark}[Important caveat]
In general,
\[
\left(\sum_{i=1}^m \lambda_i B_{i,\ell}\right)\left(\sum_{i=1}^m \lambda_i A_{i,\ell}\right)\neq \sum_{i=1}^m \lambda_i B_{i,\ell}A_{i,\ell}.
\]
Therefore the thesis must distinguish between linearly combining \emph{effective LoRA updates} and naively averaging \(A_i\) and \(B_i\) separately.
\end{remark}
